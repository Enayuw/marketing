# 智能营销中台数据链路追踪技术方案

## 一、方案架构

### 1.1 整体架构图
```
业务系统 -> 数据采集SDK -> 本地日志 -> FileBeat -> Kafka -> 数据处理 -> Doris -> 链路统计 -> 可视化展示
   ↓           ↓           ↓         ↓       ↓        ↓        ↓         ↓         ↓
多种接入    多维字段     异步写入   日志收集  消息队列  批量消费   列式存储   聚合计算   实时查询
```

### 1.2 技术选型
- **数据采集**：自研SDK + Logback + FileBeat
- **消息队列**：Kafka（利用现有）
- **数据存储**：Doris（利用现有）
- **数据处理**：Spring Boot + Kafka Consumer
- **可视化**：前端 + REST API

### 2.2 核心采集器

```java
/**
 * 链路追踪核心采集器
 */
@Slf4j
public class TraceCollector {
    
    private static final Logger TRACE_LOGGER = LoggerFactory.getLogger("TRACE_DATA");
    
    private static final ThreadLocal<TraceContext> TRACE_CONTEXT = new ThreadLocal<>();
    
    /**
     * 初始化链路上下文
     */
    public static void initTrace(String custNum, String dataSourceType) {
        TraceContext context = new TraceContext();
        context.setTraceId(generateTraceId());
        context.setCustNum(custNum);
        context.setDataSourceType(dataSourceType);
        context.setStartTime(System.currentTimeMillis());
        TRACE_CONTEXT.set(context);
    }
    
    /**
     * 记录节点执行
     */
    public static void recordNode(String nodeType, String name, String cell) {
        TraceContext context = TRACE_CONTEXT.get();
        if (context == null) {
            log.warn("Trace context not found, skipping record");
            return;
        }
        
        TraceData trace = new TraceData();
        trace.setTraceId(context.getTraceId());
        trace.setRequestId(context.getRequestId());
        trace.setSpanId(generateSpanId());
        trace.setParentSpanId(context.getCurrentSpanId());
        
        trace.setDataSourceType(context.getDataSourceType());
        trace.setNodeType(nodeType);
        trace.setDataTime(context.getDataTime());
        trace.setCell(cell);
        trace.setCustNum(context.getCustNum());
        trace.setName(name);
        
        trace.setServiceName(getServiceName());
        trace.setStatus("PROCESSING");
        trace.setStartTime(System.currentTimeMillis());
        trace.setHostName(getHostName());
        trace.setHostIp(getHostIp());
        trace.setTimestamp(System.currentTimeMillis());
        
        // 更新上下文
        context.setCurrentSpanId(trace.getSpanId());
        
        // 写入日志
        writeTraceLog(trace);
    }
    
    /**
     * 记录节点完成
     */
    public static void finishNode(boolean success, String errorMsg) {
        TraceContext context = TRACE_CONTEXT.get();
        if (context == null) return;
        
        TraceData trace = context.getCurrentTrace();
        if (trace != null) {
            trace.setEndTime(System.currentTimeMillis());
            trace.setDuration(trace.getEndTime() - trace.getStartTime());
            trace.setStatus(success ? "SUCCESS" : "FAILED");
            trace.setErrorMsg(errorMsg);
            
            // 重新写入日志（更新状态）
            writeTraceLog(trace);
        }
    }
    
    /**
     * 批量记录（用于大批量数据处理）
     */
    public static void recordBatch(String nodeType, String name, 
                                 List<String> cells, int successCount, int failCount) {
        TraceContext context = TRACE_CONTEXT.get();
        if (context == null) return;
        
        // 记录批次摘要
        TraceData batchTrace = new TraceData();
        batchTrace.setTraceId(context.getTraceId());
        batchTrace.setSpanId(generateSpanId());
        batchTrace.setNodeType(nodeType);
        batchTrace.setName(name);
        batchTrace.setCustNum(context.getCustNum());
        batchTrace.setStatus(failCount == 0 ? "SUCCESS" : "PARTIAL_SUCCESS");
        batchTrace.setStartTime(System.currentTimeMillis());
        batchTrace.setEndTime(System.currentTimeMillis());
        
        Map<String, Object> extInfo = new HashMap<>();
        extInfo.put("totalCount", cells.size());
        extInfo.put("successCount", successCount);
        extInfo.put("failCount", failCount);
        extInfo.put("batchType", "SUMMARY");
        batchTrace.setExtInfo(extInfo);
        
        writeTraceLog(batchTrace);
        
        // 采样记录单条数据（1%采样率）
        for (String cell : sampleCells(cells, 0.01)) {
            TraceData sampleTrace = new TraceData();
            sampleTrace.setTraceId(context.getTraceId());
            sampleTrace.setSpanId(generateSpanId());
            sampleTrace.setNodeType(nodeType);
            sampleTrace.setName(name);
            sampleTrace.setCell(cell);
            sampleTrace.setCustNum(context.getCustNum());
            sampleTrace.setStatus("SUCCESS");
            
            Map<String, Object> sampleExt = new HashMap<>();
            sampleExt.put("batchType", "SAMPLE");
            sampleTrace.setExtInfo(sampleExt);
            
            writeTraceLog(sampleTrace);
        }
    }
    
    /**
     * 清理上下文
     */
    public static void clearTrace() {
        TRACE_CONTEXT.remove();
    }
    
    /**
     * 写入trace日志
     */
    private static void writeTraceLog(TraceData trace) {
        try {
            TRACE_LOGGER.info(trace.toLogString());
        } catch (Exception e) {
            log.error("写入trace日志失败", e);
        }
    }
    
    /**
     * 生成TraceID
     */
    private static String generateTraceId() {
        return System.currentTimeMillis() + "_" + 
               ThreadLocalRandom.current().nextInt(1000, 9999) + "_" +
               getHostIp().replace(".", "");
    }
    
    /**
     * 生成SpanID
     */
    private static String generateSpanId() {
        return UUID.randomUUID().toString().replace("-", "").substring(0, 16);
    }
}

    private String requestId;
    private String currentSpanId;
    private String custNum;
    private String dataSourceType;
    private Long dataTime;
    private Long startTime;
    private TraceData currentTrace;
}
```

```java
/**
 * 链路追踪注解
 */
@Target({ElementType.METHOD})
@Retention(RetentionPolicy.RUNTIME)
@Documented
public @interface TraceRecord {
    
    String nodeType();           // 节点类型
    String name();               // 节点名称
    String dataSourceType() default ""; // 数据源类型
    String custNumField() default "";   // 客户编号字段名
    String cellField() default "";      // cell字段名
    boolean isBatch() default false;    // 是否批量处理
}

/**
 * 链路追踪切面
 */
@Aspect
@Component
@Slf4j
public class TraceAspect {
    
    @Around("@annotation(traceRecord)")
    public Object traceMethod(ProceedingJoinPoint joinPoint, TraceRecord traceRecord) throws Throwable {
        
        Object[] args = joinPoint.getArgs();
        
        // 提取客户编号
        String custNum = extractCustNum(args, traceRecord.custNumField());
        
        // 初始化链路（如果未初始化）
        if (TraceCollector.getCurrentTrace() == null) {
            String dataSourceType = StringUtils.isNotEmpty(traceRecord.dataSourceType()) ? 
                traceRecord.dataSourceType() : inferDataSourceType(joinPoint);
            TraceCollector.initTrace(custNum, dataSourceType);
        }
        
        // 提取cell
        String cell = extractCell(args, traceRecord.cellField());
        
        // 记录节点开始
        TraceCollector.recordNode(traceRecord.nodeType(), traceRecord.name(), cell);
        
        boolean success = false;
        String errorMsg = null;
        
        try {
            Object result = joinPoint.proceed();
            success = true;
            
            // 如果是批量处理，记录批量结果
            if (traceRecord.isBatch()) {
                recordBatchResult(args, result, traceRecord);
            }
            
            return result;
            
        } catch (Throwable e) {
            errorMsg = e.getMessage();
            throw e;
        } finally {
            TraceCollector.finishNode(success, errorMsg);
        }
    }
    
    private String extractCustNum(Object[] args, String fieldName) {
        if (StringUtils.isEmpty(fieldName)) {
            // 默认查找custNum, customerId, customerNo等字段
            return findFieldValue(args, "custNum", "customerId", "customerNo");
        }
        return findFieldValue(args, fieldName);
    }
    
    private String extractCell(Object[] args, String fieldName) {
        if (StringUtils.isEmpty(fieldName)) {
            // 默认查找手机号、身份证等字段
            return findFieldValue(args, "mobile", "phone", "idCard", "caseNo");
        }
        return findFieldValue(args, fieldName);
    }
}
```

#### 2.3.3 方式三：方法动态增强

```java
/**
 * 基于字节码增强的动态接入
 */
@Component
public class DynamicTraceEnhancer {
    
    private final Map<String, TraceConfig> methodConfigs = new ConcurrentHashMap<>();
    
    /**
     * 配置需要增强的方法
     */
    public void configureMethod(String className, String methodName, TraceConfig config) {
        String key = className + "#" + methodName;
        methodConfigs.put(key, config);
    }
    
    /**
     * 动态增强指定类的方法
     */
    public void enhanceClass(Class<?> targetClass) {
        try {
            ClassPool pool = ClassPool.getDefault();
            CtClass ctClass = pool.get(targetClass.getName());
            
            for (CtMethod method : ctClass.getDeclaredMethods()) {
                String key = targetClass.getName() + "#" + method.getName();
                TraceConfig config = methodConfigs.get(key);
                
                if (config != null) {
                    enhanceMethod(method, config);
                }
            }
            
            ctClass.toClass();
            
        } catch (Exception e) {
            log.error("动态增强类失败: {}", targetClass.getName(), e);
        }
    }
    
    private void enhanceMethod(CtMethod method, TraceConfig config) throws Exception {
        // 方法开始前插入代码
        String beforeCode = String.format("""
            String custNum = %s;
            String cell = %s;
            com.marketing.trace.TraceCollector.initTrace(custNum, "%s");
            com.marketing.trace.TraceCollector.recordNode("%s", "%s", cell);
            long startTime = System.currentTimeMillis();
            """, 
            config.getCustNumExpression(),
            config.getCellExpression(),
            config.getDataSourceType(),
            config.getNodeType(),
            config.getNodeName()
        );
        
        method.insertBefore(beforeCode);
        
        // 方法结束后插入代码
        String afterCode = """
            com.marketing.trace.TraceCollector.finishNode(true, null);
            com.marketing.trace.TraceCollector.clearTrace();
            """;
        
        method.insertAfter(afterCode);
        
        // 异常处理
        String catchCode = """
            com.marketing.trace.TraceCollector.finishNode(false, $e.getMessage());
            com.marketing.trace.TraceCollector.clearTrace();
            throw $e;
            """;
        
        CtClass exceptionType = ClassPool.getDefault().get("java.lang.Exception");
        method.addCatch(catchCode, exceptionType);
    }
}
```

### 3.2 Kafka消费处理

```java
/**
 * Trace日志消费处理器
 */
@Component
@Slf4j
public class TraceLogConsumer {
    
    @Autowired
    private DorisTraceRepository dorisRepository;
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    private final BlockingQueue<TraceDataDO> processQueue = new LinkedBlockingQueue<>(50000);
    
    @KafkaListener(topics = "trace-logs", concurrency = "5")
    public void consumeTraceLog(ConsumerRecord<String, String> record) {
        try {
            String logContent = record.value();
            
            // 解析日志内容
            TraceDataDO traceData = parseLogContent(logContent);
            if (traceData != null) {
                
                // 加入处理队列
                if (!processQueue.offer(traceData)) {
                    log.warn("处理队列已满，丢弃trace数据: {}", traceData.getTraceId());
                }
            }
            
        } catch (Exception e) {
            log.error("消费trace日志失败: {}", record.value(), e);
        }
    }
    
    /**
     * 解析日志内容为TraceData对象
     */
    private TraceDataDO parseLogContent(String logContent) {
        try {
            // 提取消息部分（去掉时间戳前缀）
            String message = extractMessage(logContent);
            String[] parts = message.split("\\|", -1);
            
            if (parts.length < 21) {
                log.warn("日志格式不正确，字段数量不足: {}", logContent);
                return null;
            }
            
            TraceDataDO traceData = new TraceDataDO();
            traceData.setTraceId(parts[0]);
            traceData.setRequestId(parts[1]);
            traceData.setSpanId(parts[2]);
            traceData.setParentSpanId(parts[3]);
            traceData.setDataSourceType(parts[4]);
            traceData.setNodeType(parts[5]);
            traceData.setDataTime(parseLong(parts[6]));
            traceData.setCell(parts[7]);
            traceData.setCustNum(parts[8]);
            traceData.setName(parts[9]);
            traceData.setServiceName(parts[10]);
            traceData.setMethodName(parts[11]);
            traceData.setStatus(parts[12]);
            traceData.setStartTime(parseLong(parts[13]));
            traceData.setEndTime(parseLong(parts[14]));
            traceData.setDuration(parseLong(parts[15]));
            traceData.setErrorMsg(parts[16]);
            traceData.setExtInfo(parts[17]);
            traceData.setHostName(parts[18]);
            traceData.setHostIp(parts[19]);
            traceData.setTimestamp(parseLong(parts[20]));
            
            return traceData;
            
        } catch (Exception e) {
            log.error("解析trace日志失败: {}", logContent, e);
            return null;
        }
    }
    
    /**
     * 批量处理和写入Doris
     */
    @Scheduled(fixedDelay = 5000) // 5秒批量处理一次
    public void batchProcessToDoris() {
        List<TraceDataDO> batch = new ArrayList<>();
        processQueue.drainTo(batch, 1000);
        
        if (batch.isEmpty()) return;
        
        try {
            // 数据清洗和转换
            List<TraceDataDO> cleanedData = cleanAndTransform(batch);
            
            // 批量写入Doris
            dorisRepository.batchInsert(cleanedData);
            
            // 更新实时统计缓存
            updateRealTimeStats(cleanedData);
            
            log.info("批量写入Doris成功，数据量: {}", cleanedData.size());
            
        } catch (Exception e) {
            log.error("批量写入Doris失败", e);
            // 重新放回队列重试
            processQueue.addAll(batch);
        }
    }
    
    /**
     * 数据清洗和转换
     */
    private List<TraceDataDO> cleanAndTransform(List<TraceDataDO> batch) {
        return batch.stream()
            .filter(this::isValidData)
            .map(this::enrichData)
            .collect(Collectors.toList());
    }
    
    private boolean isValidData(TraceDataDO data) {
        return StringUtils.isNotEmpty(data.getTraceId()) &&
               StringUtils.isNotEmpty(data.getCustNum()) &&
               StringUtils.isNotEmpty(data.getNodeType());
    }
    
    private TraceDataDO enrichData(TraceDataDO data) {
        // 补充缺失字段
        if (data.getTimestamp() == null) {
            data.setTimestamp(System.currentTimeMillis());
        }
        
        // 标准化状态
        if ("SUCCESS".equals(data.getStatus()) || "FAILED".equals(data.getStatus())) {
            // 状态正常
        } else {
            data.setStatus("UNKNOWN");
        }
        
        // 计算耗时
        if (data.getDuration() == null && data.getStartTime() != null && data.getEndTime() != null) {
            data.setDuration(data.getEndTime() - data.getStartTime());
        }
        
        return data;
    }
    
    /**
     * 更新实时统计缓存
     */
    private void updateRealTimeStats(List<TraceDataDO> data) {
        Map<String, CustomerRealTimeStat> customerStats = new HashMap<>();
        
        for (TraceDataDO trace : data) {
            String custNum = trace.getCustNum();
            CustomerRealTimeStat stat = customerStats.computeIfAbsent(custNum, 
                k -> new CustomerRealTimeStat(custNum));
            
            stat.incrementTotal();
            if ("SUCCESS".equals(trace.getStatus())) {
                stat.incrementSuccess();
            } else if ("FAILED".equals(trace.getStatus())) {
                stat.incrementFailed();
            }
            
            if (trace.getDuration() != null) {
                stat.addDuration(trace.getDuration());
            }
        }
        
        // 更新Redis缓存
        for (CustomerRealTimeStat stat : customerStats.values()) {
            String key = "realtime_stat:" + stat.getCustNum();
            redisTemplate.opsForValue().set(key, stat, 1, TimeUnit.HOURS);
        }
    }
```

```java
        Map<String, TraceNodeVO> nodeMap = new HashMap<>();
        
        for (Map<String, Object> row : results) {
            TraceNodeVO node = new TraceNodeVO();
            node.setTraceId((String) row.get("trace_id"));
            node.setSpanId((String) row.get("span_id"));
            node.setParentSpanId((String) row.get("parent_span_id"));
            node.setDataSourceType((String) row.get("data_source_type"));
            node.setNodeType((String) row.get("node_type"));
            node.setNodeName((String) row.get("name"));
            node.setCustNum((String) row.get("cust_num"));
            node.setServiceName((String) row.get("service_name"));
            node.setStatus((String) row.get("status"));
            node.setStartTime((Long) row.get("start_time"));
            node.setEndTime((Long) row.get("end_time"));
            node.setDuration((Long) row.get("duration"));
            node.setErrorMsg((String) row.get("error_msg"));
            
            nodeMap.put(node.getSpanId(), node);
        }
        
        // 构建父子关系
        List<TraceNodeVO> rootNodes = new ArrayList<>();
        for (TraceNodeVO node : nodeMap.values()) {
            if (StringUtils.isEmpty(node.getParentSpanId())) {
                rootNodes.add(node);
            } else {
                TraceNodeVO parent = nodeMap.get(node.getParentSpanId());
                if (parent != null) {
                    if (parent.getChildren() == null) {
                        parent.setChildren(new ArrayList<>());
                    }
                    parent.getChildren().add(node);
                }
            }
        }
        
        // 按时间排序
        return rootNodes.stream()
            .sorted(Comparator.comparing(TraceNodeVO::getStartTime))
            .collect(Collectors.toList());
    }
```

### 6.3 告警规则

```yaml
# trace_alerts.yml
groups:
  - name: trace_data_alerts
    rules:
      - alert: TraceDataProcessingLag
        expr: kafka_consumer_lag_sum{topic="trace-logs"} > 10000
        for: 2m
        labels:
          severity: warning
        annotations:
          summary: "Trace数据处理延迟过高"
          description: "Kafka消费延迟超过10000条消息"
          
      - alert: TraceDataLoss
        expr: increase(trace_data_received_total[5m]) > increase(trace_data_processed_total[5m]) * 1.1
        for: 3m
        labels:
          severity: critical
        annotations:
          summary: "Trace数据丢失"
          description: "5分钟内处理的数据量明显少于接收的数据量"
          
      - alert: CustomerSuccessRateDropping
        expr: customer_success_rate < 0.9
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "客户成功率下降"
          description: "客户{{ $labels.cust_num }}成功率降至{{ $value }}%"
          
      - alert: NodeProcessingTimeHigh
        expr: avg_processing_time_seconds > 30
        for: 2m
        labels:
          severity: warning
        annotations:
          summary: "节点处理时间过长"
          description: "节点{{ $labels.node_name }}平均处理时间超过30秒"

  - name: infrastructure_alerts
    rules:
      - alert: DorisQuerySlow
        expr: doris_query_duration_seconds_p95 > 5
        for: 1m
        labels:
          severity: warning
        annotations:
          summary: "Doris查询响应慢"
          description: "Doris查询95分位响应时间超过5秒"
          
      - alert: RedisMemoryUsageHigh
        expr: redis_memory_used_bytes / redis_memory_max_bytes > 0.8
        for: 2m
        labels:
          severity: warning
        annotations:
          summary: "Redis内存使用率过高"
```
